  \def\cyld{#3}\def\cylh{#2}

  \fill[top color=gray!90!,bottom color=gray!2,middle color=gray!30,shading=axis,opacity=0.25]
    (0,\cylh) circle (\cyld cm and 0.5cm);
  \draw (-\cyld,\cylh) -- (-\cyld,0) arc (180:360:\cyld cm and 0.5cm) -- (\cyld,\cylh) ++ (-\cyld,0) circle (\cyld cm and 0.5cm);
  \draw[densely dashed] (-\cyld,0) arc (180:0:\cyld cm and 0.5cm);
  \draw[<->,thin] (\cyld+0.35,0) -- (\cyld+0.35,\cylh) node[midway,right,font=\tiny]{\cylh};
  \draw[<->,thin] (-\cyld,-0.4) -- (\cyld,-0.4) node[midway,below,font=\tiny]{$2\cyld$};

\begin{frame}[t,shrink=10]
\NotesTitle{Surface area}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\KeyIdea{Surface area is the total area of all the outside faces of a 3D shape. A good method is to imagine the net, find the area of each face, then add them all together.}
\SectionTitle{Steps}
\StepLine{1}{Identify every outside face.}
\StepLine{2}{Find the area of each face.}
\StepLine{3}{Multiply pairs of congruent faces when possible.}
\StepLine{4}{Add the areas and give square units.}

\CommonMistake{Missing one face. For example, a closed cuboid has 6 faces and a closed cylinder has 2 circles plus 1 curved surface.}
\end{column}
\begin{column}{0.48\textwidth}
\WorkedExample{Cube}{%
\DrawCuboid[0.5]{4}{4}{4}\\[2pt]
A cube has side length 4 cm.\\[2pt]
Each face: $4\times 4 = 16$ cm$^2$\\
6 faces: $6\times 16 = 96$ cm$^2$\\[2pt]
\textbf{Surface area = 96 cm$^2$}}

\WorkedExample{Cuboid}{%
\DrawCuboid[0.45]{8}{3}{2}\\[2pt]
A box is 8 cm by 3 cm by 2 cm.\\[2pt]
Top \& bottom: $2(8\times 3)=48$\\
Front \& back: $2(8\times 2)=32$\\
Left \& right: $2(3\times 2)=12$\\
Total: $48+32+12=92$ cm$^2$\\[2pt]
\textbf{Surface area = 92 cm$^2$}}
\end{column}
\end{columns}
\GreenDivider
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\WorkedExample{Cylinder}{%
\DrawCylinder[0.55]{10}{3}\\[2pt]
A cylinder has radius 3 cm and height 10 cm.\\[2pt]
Two circles: $2\pi r^2 = 2\pi(3^2)=18\pi$\\
Curved surface: $2\pi rh = 2\pi(3)(10)=60\pi$\\
Total: $18\pi + 60\pi = 78\pi$ cm$^2$\\[2pt]
\textbf{Surface area = $78\pi$\,cm$^2$}}
\end{column}
\begin{column}{0.48\textwidth}
\TryThese{\begin{enumerate}
\item A cube has side length 6 cm. Find its surface area.
\item A cuboid is 10 cm by 4 cm by 2 cm. Find its surface area.
\item A cylinder has radius 4 cm and height 9 cm. Write its total surface area in terms of $\pi$.
\end{enumerate}}
\end{column}
