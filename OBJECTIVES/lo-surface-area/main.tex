\NotesTitle{Surface area}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\KeyIdea{Surface area is the total area of all the outside faces of a 3D shape. For a net, you find the area of each shape in the net and add them together.}
\SectionTitle{Steps}
\StepLine{1}{Identify each face of the 3D shape.}
\StepLine{2}{Find the area of each face.}
\StepLine{3}{Add all the areas together.}
\StepLine{4}{Write the answer with square units (cm$^2$, m$^2$, etc.).}

\CommonMistake{Forgetting to include both ends. A cylinder has two circular faces (top and bottom), not one. A cuboid has 6 faces, not 5.}
\end{column}
\begin{column}{0.48\textwidth}
\WorkedExample{Cube}{A cube has side length 4 cm.\\[2pt]
Each face is a square: $4 \times 4 = 16$ cm$^2$.\\
A cube has 6 faces.\\
$6 \times 16 = 96$ cm$^2$.\par\vspace{0.1em}
Answer: \textbf{96 cm$^2$}}

\WorkedExample{Cuboid}{A box is 8 cm long, 3 cm wide, 2 cm high.\\[2pt]
Top \& bottom: $2 \times (8 \times 3) = 48$ cm$^2$\\
Sides: $2 \times (8 \times 2) = 32$ cm$^2$\\
Ends: $2 \times (3 \times 2) = 12$ cm$^2$\\
Total: $48 + 32 + 12 = 92$ cm$^2$.\par\vspace{0.1em}
Answer: \textbf{92 cm$^2$}}
\end{column}
\end{columns}
