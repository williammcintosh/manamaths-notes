\KeyIdea{The derivative $f'(x)$ is found using differentiation by first principles:
\[
f'(x) = \lim_{h \to 0} \frac{f(x+h) - f(x)}{h}
\]
This gives the gradient function of $f(x)$.}

\SectionTitle{Steps}
\StepLine{1}{Write $f(x+h)$ by replacing $x$ with $x+h$.}
\StepLine{2}{Expand and simplify $f(x+h) - f(x)$.}
\StepLine{3}{Divide by $h$: $\frac{f(x+h) - f(x)}{h}$.}
\StepLine{4}{Take the limit as $h \to 0$ (the terms with $h$ disappear).}
\end{column}
\begin{column}{0.48\textwidth}
\SectionTitle{Quick checks}
\begin{itemize}
\item For $f(x) = x^2$: $f'(x) = 2x$
\item For $f(x) = 4x$: $f'(x) = 4$
\item For $f(x) = x^2 + 5$: $f'(x) = 2x$
\item Constants differentiate to zero
\end{itemize}
\end{column}
\end{columns}

\WorkedExample{Example 1: $f(x) = x^2$}{
$f(x+h) = (x+h)^2 = x^2 + 2xh + h^2$\\
$f(x+h) - f(x) = (x^2 + 2xh + h^2) - x^2 = 2xh + h^2$\\
$\frac{f(x+h) - f(x)}{h} = \frac{2xh + h^2}{h} = 2x + h$\\
As $h \to 0$: $f'(x) = 2x$
}
\WorkedExample{Example 2: $f(x) = 3x^2 + 2x$}{
$f(x+h) = 3(x+h)^2 + 2(x+h) = 3x^2 + 6xh + 3h^2 + 2x + 2h$\\
$f(x+h) - f(x) = 6xh + 3h^2 + 2h = h(6x + 3h + 2)$\\
$\frac{f(x+h) - f(x)}{h} = 6x + 3h + 2$\\
As $h \to 0$: $f'(x) = 6x + 2$
}
\end{column}
\begin{column}{0.48\textwidth}
\WorkedExample{Example 3: $f(x) = x^3$}{
$f(x+h) = (x+h)^3 = x^3 + 3x^2h + 3xh^2 + h^3$\\
$f(x+h) - f(x) = 3x^2h + 3xh^2 + h^3 = h(3x^2 + 3xh + h^2)$\\
$\frac{f(x+h) - f(x)}{h} = 3x^2 + 3xh + h^2$\\
As $h \to 0$: $f'(x) = 3x^2$
}
\TryThese{\begin{enumerate}
  \item $f(x) = 2x^2$ — find $f'(x)$ by first principles
  \item $f(x) = x^2 + 4x$ — find $f'(x)$ by first principles
  \item $f(x) = 5x^2 - 3x + 2$ — find $f'(x)$ by first principles
\end{enumerate}}
\CommonMistake{When expanding $(x+h)^3$, students often miss the $3xh^2$ term. Write it out term by term: $(x+h)^3 = (x+h)(x^2 + 2xh + h^2)$.}
\end{column}
